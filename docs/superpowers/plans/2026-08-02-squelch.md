# squelch Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A terminal application that plays a LiveATC.net air traffic control feed and transcribes it live in a 48×42 side pane.

**Architecture:** One process, one network connection. A single `ffmpeg` subprocess decodes the stream to 16 kHz mono PCM; those frames fan out to a `sounddevice` playback stream and to a WebRTC VAD segmenter. The segmenter emits whole radio transmissions, which a faster-whisper worker thread transcribes and posts to a Textual UI. Modules below the UI layer never import Textual.

**Tech Stack:** Python 3.14, `uv`, Textual 8.2, faster-whisper 1.2 (CTranslate2), `webrtcvad-wheels`, `sounddevice`, external `ffmpeg` 7.1.

**Spec:** `docs/superpowers/specs/2026-08-02-squelch-design.md`

## Global Constraints

- Target render size is exactly **48 columns × 42 rows**. No rendered line may exceed 48 cells.
- Audio format throughout the pipeline is **16 kHz, mono, signed 16-bit little-endian** (`s16le`).
- VAD frame size is **20 ms = 320 samples = 640 bytes**. Every `Segmenter.feed()` call takes exactly 640 bytes.
- **WebRTC VAD is stateful and must always be gated by frame energy.** Verified on this machine: after 25 loud frames, `Vad.is_speech()` reports the next 6 frames of pure digital silence (120 ms) as speech. Speech detection is therefore `rms_dbfs(frame) > gate_dbfs and vad.is_speech(frame)`, with `gate_dbfs = -70.0`, which was measured to remove the artifact entirely while still detecting every genuine speech frame. Never call `is_speech` ungated.
- `feeds.py`, `vad.py`, `asr.py`, `audio.py`, `pipeline.py`, and `config.py` **must not import `textual`**.
- No test may access the network or load a real ASR model, except the explicitly-marked smoke test in Task 9.
- Reconnect backoff is **1 s doubling to a 30 s cap**.
- Package layout is `src/squelch/`, tests in `tests/`.
- The environment is already verified: `uv` 0.11.14, Python 3.14.6, `ffmpeg` 7.1.1, and all native wheels (`ctranslate2` 4.8.1, `webrtcvad-wheels` 2.0.14, `sounddevice` 0.5.5) install and import on arm64 macOS.

```python
"""Configuration loading for squelch."""

from __future__ import annotations

import tomllib
from dataclasses import dataclass, fields
from pathlib import Path

DEFAULT_PATH = Path.home() / ".config" / "squelch" / "squelch.toml"
DEFAULT_MODEL = "jacktol/whisper-medium.en-fine-tuned-for-ATC-faster-whisper"


@dataclass(frozen=True)
class Config:
    icao: str = "KPDX"
    feed: str = "kpdx_twr"
    model: str = DEFAULT_MODEL
    compute_type: str = "int8"
    vad_aggressiveness: int = 2


def load_config(path: Path | None = None) -> Config:
    """Load config from TOML, falling back to defaults.

    Unknown keys are ignored so a newer config file does not crash an
    older binary.
    """
    path = DEFAULT_PATH if path is None else path
    if not path.exists():
        return Config()
    data = tomllib.loads(path.read_text())
    known = {f.name for f in fields(Config)}
    return Config(**{k: v for k, v in data.items() if k in known})
```

```python
"""Discovery and resolution of LiveATC.net feeds."""

from __future__ import annotations

import re
import urllib.request
from collections.abc import Callable
from dataclasses import dataclass

Fetcher = Callable[[str], str]

BASE = "https://www.liveatc.net"
USER_AGENT = "squelch/0.1 (personal listener)"

_PLS_FILE_RE = re.compile(r"^\s*File\d+\s*=\s*(\S+)\s*$", re.MULTILINE)
_FEED_LINK_RE = re.compile(r"/play/([a-z0-9_\-]+)\.pls", re.IGNORECASE)


class FeedError(Exception):
    """A feed could not be discovered or resolved."""


@dataclass(frozen=True)
class Feed:
    ident: str
    label: str


def pls_url(ident: str) -> str:
    return f"{BASE}/play/{ident}.pls"


def airport_url(icao: str) -> str:
    return f"{BASE}/search/?icao={icao.lower()}"


def parse_pls(text: str) -> str:
    """Return the first stream URL in a PLS playlist."""
    match = _PLS_FILE_RE.search(text)
    if not match:
        raise FeedError("no File entry in playlist")
    return match.group(1)


def parse_airport_page(html: str) -> list[Feed]:
    """Extract feeds from a LiveATC airport search page.

    Feeds are anchored on /play/<ident>.pls links. The label is taken from
    the nearest preceding heading-ish text, falling back to a title-cased
    ident so a label is never empty.
    """
    feeds: list[Feed] = []
    seen: set[str] = set()
    for match in _FEED_LINK_RE.finditer(html):
        ident = match.group(1).lower()
        if ident in seen:
            continue
        seen.add(ident)
        feeds.append(Feed(ident=ident, label=_label_for(html, match.start(), ident)))
    if not feeds:
        raise FeedError("no feeds found on page")
    return feeds


def _label_for(html: str, pos: int, ident: str) -> str:
    window = html[max(0, pos - 1500) : pos]
    candidates = re.findall(r"<strong>([^<]{3,60})</strong>", window)
    if candidates:
        return _clean(candidates[-1])
    return ident.replace("_", " ").upper()


def _clean(text: str) -> str:
    return re.sub(r"\s+", " ", text).strip()


def _http_get(url: str) -> str:
    request = urllib.request.Request(url, headers={"User-Agent": USER_AGENT})
    with urllib.request.urlopen(request, timeout=15) as response:
        return response.read().decode("utf-8", errors="replace")


def fetch_feeds(icao: str, fetch: Fetcher = _http_get) -> list[Feed]:
    return parse_airport_page(fetch(airport_url(icao)))


def resolve_stream(ident: str, fetch: Fetcher = _http_get) -> str:
    return parse_pls(fetch(pls_url(ident)))
```

```python
"""Segment a PCM stream into discrete radio transmissions.

ATC audio is squelch-gated: hard silence between transmissions rather than
continuous room tone. That makes voice activity detection easy and lets us
transcribe whole transmissions as units instead of fixed-size windows.
"""

from __future__ import annotations

import array
import math
from dataclasses import dataclass

import webrtcvad

SAMPLE_RATE = 16_000
FRAME_MS = 20
FRAME_SAMPLES = SAMPLE_RATE * FRAME_MS // 1000  # 320
FRAME_BYTES = FRAME_SAMPLES * 2                 # 640
SILENCE_FLOOR_DBFS = -96.0
GATE_DBFS = -70.0


def rms_dbfs(pcm: bytes) -> float:
    """Frame energy in dBFS, floored at SILENCE_FLOOR_DBFS."""
    samples = array.array("h")
    samples.frombytes(pcm)
    if not samples:
        return SILENCE_FLOOR_DBFS
    mean_square = sum(s * s for s in samples) / len(samples)
    if mean_square <= 0:
        return SILENCE_FLOOR_DBFS
    return max(
        SILENCE_FLOOR_DBFS, 20 * math.log10(math.sqrt(mean_square) / 32768.0)
    )


@dataclass(frozen=True)
class Utterance:
    start_s: float
    end_s: float
    pcm: bytes

    @property
    def duration_s(self) -> float:
        return self.end_s - self.start_s


class Segmenter:
    """Turn 20 ms PCM frames into utterances.

    An utterance opens on the first speech frame and closes after
    `hangover_ms` of continuous silence, so a mid-sentence pause does not
    split one transmission into two. Utterances shorter than `min_ms` are
    discarded as squelch clicks; one running longer than `max_ms` is
    force-flushed so a stuck carrier cannot buffer forever.
    """

    def __init__(
        self,
        *,
        aggressiveness: int = 2,
        hangover_ms: int = 600,
        min_ms: int = 400,
        max_ms: int = 20_000,
        gate_dbfs: float = GATE_DBFS,
    ) -> None:
        self._vad = webrtcvad.Vad(aggressiveness)
        self._gate_dbfs = gate_dbfs
        self._hangover_frames = max(1, hangover_ms // FRAME_MS)
        self._min_frames = max(1, min_ms // FRAME_MS)
        self._max_frames = max(1, max_ms // FRAME_MS)
        self._buffer: list[bytes] = []
        self._silence_run = 0
        self._frame_index = 0
        self._start_frame = 0

    def feed(self, frame: bytes) -> Utterance | None:
        if len(frame) != FRAME_BYTES:
            raise ValueError(f"expected {FRAME_BYTES} bytes, got {len(frame)}")

        index = self._frame_index
        self._frame_index += 1
        is_speech = self._is_speech(frame)

        if is_speech:
            if not self._buffer:
                self._start_frame = index
            self._buffer.append(frame)
            self._silence_run = 0
            if len(self._buffer) >= self._max_frames:
                return self._close(index + 1, enforce_min=False)
            return None

        if not self._buffer:
            return None

        # Silence while an utterance is open: keep it as possible interior
        # pause until the hangover expires.
        self._buffer.append(frame)
        self._silence_run += 1
        if self._silence_run >= self._hangover_frames:
            end = index + 1 - self._silence_run
            return self._close(end, enforce_min=True)
        return None

    def _is_speech(self, frame: bytes) -> bool:
        """Energy-gated VAD.

        WebRTC's VAD adapts to recent input, so immediately after loud audio
        it reports pure silence as speech for a hundred milliseconds or so.
        Gating on frame energy first removes that tail, and suits
        squelch-gated ATC audio where the silence is digitally clean.
        """
        if rms_dbfs(frame) <= self._gate_dbfs:
            return False
        return self._vad.is_speech(frame, SAMPLE_RATE)

    def flush(self) -> Utterance | None:
        if not self._buffer:
            return None
        return self._close(self._frame_index - self._silence_run, enforce_min=False)

    def _close(self, end_frame: int, *, enforce_min: bool) -> Utterance | None:
        voiced = self._buffer[: len(self._buffer) - self._silence_run] or self._buffer
        buffered = list(voiced)
        self._buffer.clear()
        self._silence_run = 0
        if enforce_min and len(buffered) < self._min_frames:
            return None
        return Utterance(
            start_s=_seconds(self._start_frame),
```

```python
"""Stream acquisition and playback.

One ffmpeg process decodes the feed to 16 kHz mono PCM. Those frames feed
both the speaker and the transcription pipeline, so the two can never drift
apart.
"""

from __future__ import annotations

import subprocess
import time
from collections.abc import Callable, Iterator

from squelch.vad import FRAME_BYTES, SAMPLE_RATE

Spawner = Callable[[str], subprocess.Popen]

READ_SIZE = FRAME_BYTES * 4


def ffmpeg_command(url: str) -> list[str]:
    return [
        "ffmpeg",
        "-hide_banner",
        "-loglevel", "error",
        "-nostdin",
        "-i", url,
        "-vn",
        "-ac", "1",
        "-ar", str(SAMPLE_RATE),
        "-f", "s16le",
        "-",
    ]


def backoff_delays(cap: float = 30.0) -> Iterator[float]:
    """1, 2, 4, 8, ... capped. Feed dropouts are routine, not exceptional."""
    delay = 1.0
    while True:
        yield min(delay, cap)
        delay *= 2


def _spawn_ffmpeg(url: str) -> subprocess.Popen:
    return subprocess.Popen(
        ffmpeg_command(url),
        stdout=subprocess.PIPE,
        stderr=subprocess.DEVNULL,
    )


class StreamReader:
    """Yield fixed-size PCM frames, respawning ffmpeg on failure."""

    def __init__(
        self,
        url: str,
        *,
        spawn: Spawner = _spawn_ffmpeg,
        sleep: Callable[[float], None] = time.sleep,
    ) -> None:
        self._url = url
        self._spawn = spawn
        self._sleep = sleep
        self._stopped = False
        self._process: subprocess.Popen | None = None
        self.reconnecting = False

    def frames(self) -> Iterator[bytes]:
        delays = backoff_delays()
        buffer = b""
        while not self._stopped:
            self._process = self._spawn(self._url)
            self.reconnecting = False
            while not self._stopped:
                chunk = self._process.stdout.read(READ_SIZE)
                if not chunk:
                    break
                buffer += chunk
                while len(buffer) >= FRAME_BYTES:
                    yield buffer[:FRAME_BYTES]
                    buffer = buffer[FRAME_BYTES:]
                    if self._stopped:
                        break
            self._kill()
            if self._stopped:
                return
            self.reconnecting = True
            self._sleep(next(delays))

    def stop(self) -> None:
        self._stopped = True
        self._kill()

    def _kill(self) -> None:
        if self._process is None:
            return
        try:
            self._process.kill()
            self._process.wait(timeout=2)
        except Exception:
            pass
        self._process = None


class Player:
    """Speaker output. Muting stops audio but never stops transcription."""

    def __init__(self) -> None:
        self.muted = False
        self._stream = None

    def start(self) -> None:
        import sounddevice

        self._stream = sounddevice.RawOutputStream(
            samplerate=SAMPLE_RATE, channels=1, dtype="int16"
        )
        self._stream.start()

    def write(self, pcm: bytes) -> None:
        if self._stream is None or self.muted:
            return
        try:
            self._stream.write(pcm)
        except Exception:
            pass  # An underrun must never take down the pipeline.

    def toggle_mute(self) -> bool:
        self.muted = not self.muted
        return self.muted

    def stop(self) -> None:
        if self._stream is not None:
            self._stream.stop()
            self._stream.close()
            self._stream = None
```

```python
"""Speech recognition using an ATC-fine-tuned Whisper model.

Stock Whisper is not adequate for ATC audio: clipped VHF bandwidth, heavy
accent variety, and rapid alphanumeric callsigns push it into producing
fluent, confident, wrong text. The fine-tuned CTranslate2 weights are
materially better on callsigns and standard phraseology.
"""

from __future__ import annotations

from collections.abc import Callable
from dataclasses import dataclass
from typing import Any, Protocol

import numpy as np

Loader = Callable[..., Any]


@dataclass(frozen=True)
class Transcript:
    text: str
    avg_logprob: float


class Transcriber(Protocol):
    def transcribe(self, pcm: bytes) -> Transcript: ...


def pcm_to_float32(pcm: bytes) -> np.ndarray:
    return np.frombuffer(pcm, dtype="<i2").astype(np.float32) / 32768.0


def _default_loader(model: str, *, compute_type: str) -> Any:
    from faster_whisper import WhisperModel

    return WhisperModel(model, device="cpu", compute_type=compute_type)


class WhisperTranscriber:
    def __init__(
        self,
        model: str,
        *,
        compute_type: str = "int8",
        loader: Loader | None = None,
    ) -> None:
        self._model_id = model
        self._compute_type = compute_type
        self._loader = loader or _default_loader
        self._model: Any = None

    @property
    def ready(self) -> bool:
        return self._model is not None

    def load(self) -> None:
        """Download and load weights. Slow; call off the UI thread."""
        self._model = self._loader(self._model_id, compute_type=self._compute_type)

    def transcribe(self, pcm: bytes) -> Transcript:
        if self._model is None:
            raise RuntimeError("transcriber not loaded")
        segments, _info = self._model.transcribe(
            pcm_to_float32(pcm),
            language="en",
            beam_size=5,
            vad_filter=False,  # already segmented upstream
            condition_on_previous_text=False,  # stops hallucination loops
        )
        segments = list(segments)
        if not segments:
            return Transcript(text="", avg_logprob=0.0)
        text = " ".join(s.text.strip() for s in segments).strip()
        avg = sum(s.avg_logprob for s in segments) / len(segments)
        return Transcript(text=text, avg_logprob=avg)
```

```python
"""Wire the stream, segmenter, and transcriber together.

Owns two threads: the caller's thread runs the audio loop, and a single
worker runs inference. Inference is deliberately not allowed to apply
backpressure to audio — if the model falls behind, work is dropped and the
gap is reported, because stale text presented as live is worse than a gap.
"""

from __future__ import annotations

import queue
import threading
from collections.abc import Callable
from dataclasses import dataclass
from datetime import datetime

from squelch.vad import rms_dbfs


@dataclass(frozen=True)
class Block:
    started_at: datetime
    text: str


@dataclass
class Events:
    on_level: Callable[[float], None]
    on_pending: Callable[[], None]
    on_block: Callable[[Block], None]
    on_state: Callable[[str], None]
    on_dropped: Callable[[], None]


_SENTINEL = object()


class Pipeline:
    def __init__(
        self,
        reader,
        segmenter,
        transcriber,
        player,
        events: Events,
        *,
        now: Callable[[], datetime] = datetime.now,
        queue_size: int = 8,
    ) -> None:
        self._reader = reader
        self._segmenter = segmenter
        self._transcriber = transcriber
        self._player = player
        self._events = events
        self._now = now
        self._queue: queue.Queue = queue.Queue(maxsize=queue_size)
        self._stopped = False
        self._last_state = ""

    def run(self) -> None:
        worker = threading.Thread(target=self._worker, daemon=True)
        worker.start()
        try:
            for frame in self._reader.frames():
                if self._stopped:
                    break
                self._player.write(frame)
                self._events.on_level(rms_dbfs(frame))
                self._report_state(
                    "reconnecting" if self._reader.reconnecting else "live"
                )
                utterance = self._segmenter.feed(frame)
                if utterance is not None:
                    self._events.on_pending()
                    self._submit(utterance)
        finally:
            self._queue.put(_SENTINEL)
            worker.join(timeout=30)

    def stop(self) -> None:
        self._stopped = True
        self._reader.stop()

    def _report_state(self, state: str) -> None:
        if state != self._last_state:
            self._last_state = state
            self._events.on_state(state)

    def _submit(self, utterance) -> None:
        try:
            self._queue.put_nowait((self._now(), utterance))
        except queue.Full:
            try:
                self._queue.get_nowait()  # drop oldest
            except queue.Empty:
                pass
            self._events.on_dropped()
            try:
                self._queue.put_nowait((self._now(), utterance))
            except queue.Full:
                pass

    def _worker(self) -> None:
        while True:
            item = self._queue.get()
            if item is _SENTINEL:
                return
            started_at, utterance = item
            try:
                transcript = self._transcriber.transcribe(utterance.pcm)
            except Exception:
                continue
            if transcript.text.strip():
                self._events.on_block(
                    Block(started_at=started_at, text=transcript.text.strip())
                )
```

```python
"""Textual UI for squelch, sized for a 48-column side pane."""

from __future__ import annotations

from datetime import datetime

from textual.app import App, ComposeResult
from textual.containers import VerticalScroll
from textual.widgets import Label, Static

from squelch.config import Config
from squelch.pipeline import Block

BLOCKS = "▁▂▃▄▅▆▇█"
FLOOR_DBFS = -60.0


def meter_bar(dbfs: float, width: int = 20) -> str:
    """Render a level as block characters. -60 dBFS and below reads as idle."""
    fraction = max(0.0, min(1.0, (dbfs - FLOOR_DBFS) / -FLOOR_DBFS))
    filled = int(fraction * len(BLOCKS))
    index = max(0, min(len(BLOCKS) - 1, filled))
    if fraction <= 0.0:
        return BLOCKS[0] * width
    return BLOCKS[index] * width


class SquelchApp(App):
    CSS_PATH = "squelch.tcss"
    BINDINGS = [
        ("space", "toggle_mute", "Mute"),
        ("q", "quit", "Quit"),
    ]

    def __init__(
        self, config: Config, *, feed_label: str, frequency: str = ""
    ) -> None:
        super().__init__()
        self._config = config
        self._feed_label = feed_label
        self._frequency = frequency
        self._state = "LIVE"
        self._muted = False
        self._level = FLOOR_DBFS
        self._pending: Static | None = None
        self.on_mute_change = lambda muted: None

    def compose(self) -> ComposeResult:
        yield Static(id="header")
        yield Static(id="meter")
        yield VerticalScroll(id="transcript")
        yield Static(id="footer")

    def on_mount(self) -> None:
        self._refresh_chrome()

    # -- rendering ---------------------------------------------------

    def _refresh_chrome(self) -> None:
        left = f" {self._feed_label}  {self._frequency}".rstrip()
        right = f"● {self._state} "
        # Hard-truncate rather than pad-and-hope: the 48-column budget is the
        # premise of the layout, and the state text changes width at runtime.
        room = 48 - len(right)
        self.query_one("#header", Static).update(left[:room].ljust(room) + right)
        bar = meter_bar(self._level, width=20)
        clock = datetime.now().strftime("%H:%M:%S")
        self.query_one("#meter", Static).update(
            f" {bar} {self._level:6.1f}dB  {clock}"
        )
        hint = "MUTED" if self._muted else "mute"
        self.query_one("#footer", Static).update(
            f" [space] {hint}  [f] feeds  [q] quit"
        )

    # -- public API driven by the pipeline ---------------------------

    def add_block(self, block: Block) -> None:
        self._clear_pending()
        stamp = block.started_at.strftime("%H:%M:%S")
        widget = Static(f"{stamp}\n  {block.text}", classes="block")
        self.query_one("#transcript", VerticalScroll).mount(widget)
        self.query_one("#transcript", VerticalScroll).scroll_end(animate=False)

    def mark_pending(self) -> None:
        if self._pending is not None:
            return
        self._pending = Static("░ transcribing…", classes="block")
        self.query_one("#transcript", VerticalScroll).mount(self._pending)

    def mark_dropped(self) -> None:
        self._clear_pending()
        self.query_one("#transcript", VerticalScroll).mount(
            Static("⚠ dropped (model behind)", classes="block")
        )

    def _clear_pending(self) -> None:
        if self._pending is not None:
            self._pending.remove()
            self._pending = None

    def set_level(self, dbfs: float) -> None:
        self._level = dbfs
        self._refresh_chrome()

    def set_state(self, state: str) -> None:
        self._state = "RECONNECTING" if state == "reconnecting" else "LIVE"
        self._refresh_chrome()

    # -- actions -----------------------------------------------------

    def action_toggle_mute(self) -> None:
        self._muted = not self._muted
        self.on_mute_change(self._muted)
        self._refresh_chrome()
```

```python
"""Command line entry point."""

from __future__ import annotations

import argparse
import sys
import threading
from collections.abc import Callable
from dataclasses import replace

from squelch.app import SquelchApp
from squelch.asr import Transcript, WhisperTranscriber
from squelch.audio import Player, StreamReader
from squelch.config import Config, load_config
from squelch.feeds import fetch_feeds, resolve_stream
from squelch.pipeline import Events, Pipeline
from squelch.vad import Segmenter


def parse_args(argv: list[str] | None = None) -> argparse.Namespace:
    parser = argparse.ArgumentParser(prog="squelch", description=__doc__)
    parser.add_argument("--feed", help="LiveATC feed ident, e.g. kpdx_twr")
    parser.add_argument("--icao", help="Airport ICAO for the feed picker")
    return parser.parse_args(argv)


def try_start(
    start: Callable[[], None], on_error: Callable[[str], None], message: str
) -> bool:
    """Run a startup step, degrading rather than crashing.

    A missing model or unavailable audio device should cost you one feature,
    not the whole application.
    """
    try:
        start()
    except Exception:
        on_error(message)
        return False
    return True


class _NullTranscriber:
    """Stands in when the model will not load, so audio still plays."""

    def transcribe(self, pcm: bytes) -> Transcript:
        return Transcript(text="", avg_logprob=0.0)


def main(argv: list[str] | None = None) -> int:
    args = parse_args(argv)
    config = load_config()
    if args.feed:
        config = replace(config, feed=args.feed)
    if args.icao:
        config = replace(config, icao=args.icao)

    app = SquelchApp(
        config,
        feed_label=config.feed,
        feed_loader=lambda: fetch_feeds(config.icao),
    )

    player = Player()
    transcriber = WhisperTranscriber(config.model, compute_type=config.compute_type)
    app.on_mute_change = lambda muted: setattr(player, "muted", muted)

    events = Events(
        on_level=lambda d: app.call_from_thread(app.set_level, d),
        on_pending=lambda: app.call_from_thread(app.mark_pending),
        on_block=lambda b: app.call_from_thread(app.add_block, b),
        on_state=lambda s: app.call_from_thread(app.set_state, s),
        on_dropped=lambda: app.call_from_thread(app.mark_dropped),
    )

    def warn(message: str) -> None:
        app.call_from_thread(app.warn, message)

    def run_pipeline() -> None:
        active = transcriber
        if not try_start(
            transcriber.load, warn, "model unavailable — audio only"
        ):
            active = _NullTranscriber()
        try_start(player.start, warn, "no audio device — transcribing only")

        try:
            url = resolve_stream(config.feed)
        except Exception:
            warn(f"could not resolve feed {config.feed}")
            return

        Pipeline(
            StreamReader(url),
            Segmenter(aggressiveness=config.vad_aggressiveness),
            active,
            player,
            events,
        ).run()

    threading.Thread(target=run_pipeline, daemon=True).start()
    app.run()
    player.stop()
    return 0


if __name__ == "__main__":
    sys.exit(main())
```

## Known limitations in v1

Recorded deliberately, per the spec's non-goals:

- Selecting a feed in the picker does not restart the stream.
- No transcript logging to disk, no in-pane search or scrollback.
- No speaker labelling — a LiveATC feed is one mixed channel with no way to attribute a transmission.
- CTranslate2 has no Metal backend, so inference is CPU-bound.
